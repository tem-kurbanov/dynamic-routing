cmake_minimum_required(VERSION 3.10)
project(mas_planner)

# Set C++ standard
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

# Set build type to Release for better performance
set(CMAKE_BUILD_TYPE Release)

# Find required packages
find_package(MPI REQUIRED)
find_package(OpenMP REQUIRED)

# Set SUMO home from environment
set(SUMO_HOME $ENV{SUMO_HOME})
if(NOT SUMO_HOME)
    message(FATAL_ERROR "SUMO_HOME environment variable not set!")
endif()

# XGBoost path (can be set via command line: -DXGBOOST_ROOT=/path/to/xgboost)
if(NOT XGBOOST_ROOT)
    message(FATAL_ERROR "XGBOOST_ROOT not set! Please specify with -DXGBOOST_ROOT=/path/to/xgboost")
endif()
find_library(XGBOOST_LIB xgboost PATHS ${XGBOOST_ROOT}/lib NO_DEFAULT_PATH)
if(NOT XGBOOST_LIB)
    message(FATAL_ERROR "XGBoost library not found in ${XGBOOST_ROOT}/lib")
endif()

# Include directories
include_directories(
    ${MPI_CXX_INCLUDE_PATH}
    ${SUMO_HOME}/src
    ${XGBOOST_ROOT}/include
    ${CMAKE_CURRENT_SOURCE_DIR}
)

# Link directories
link_directories(
    ${SUMO_HOME}/bin
    ${XGBOOST_ROOT}/lib
)

# Add executable
add_executable(mas_planner
    main.cpp
    Router.cpp
    Graph.cpp
    Node.cpp
    Edge.cpp
    CoverGraph.cpp
    CoverNode.cpp
    CoverEdge.cpp
    TrafficEstimator.cpp
    SpeedEstimator.cpp
    FuelEstimator.cpp
)

# Link libraries
target_link_libraries(mas_planner
    ${MPI_CXX_LIBRARIES}
    sumocpp
    tracicpp
    ${XGBOOST_LIB}
    OpenMP::OpenMP_CXX
)

# Set compiler flags
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU")
    target_compile_options(mas_planner PRIVATE
        -march=native
        -fopenmp
    )
endif()
